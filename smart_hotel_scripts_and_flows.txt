Rule 1 — Auto-Calculate Booking Price
  Table  : x_1939650_smart_0_bookings
  When   : Before | Insert & Update
  Fields : room, check_in_date, check_out_date
────────────────────────────────────────────────────────────────────────────────

(function executeRule(current, previous) {
    if (current.room && current.check_in_date && current.check_out_date) {
        var room = new GlideRecord('x_1939650_smart_0_room');
        if (room.get(current.room)) {
            var checkIn  = new GlideDateTime(current.check_in_date.toString());
            var checkOut = new GlideDateTime(current.check_out_date.toString());
            var ms = checkOut.getNumericValue() - checkIn.getNumericValue();
            var nights = Math.ceil(ms / 86400000);
            if (nights < 1) nights = 1;
            current.total_price = parseFloat(room.price_per_night) * nights;
        }
    }
})(current, previous);

Rule 5 — Auto-Assign Urgent Incidents
  Table     : x_1939650_smart_0_guest_incidents
  When      : After | Insert
  Condition : priority is urgent
────────────────────────────────────────────────────────────────────────────────

(function executeRule(current, previous) {
    var staff = new GlideRecord('sys_user');
    staff.addActiveQuery();
    staff.addQuery('roles', 'x_1939650_smart_0.manager');
    staff.setLimit(1);
    staff.query();
    if (staff.next()) {
        var inc = new GlideRecord('x_1939650_smart_0_guest_incidents');
        if (inc.get(current.sys_id)) {
            inc.assigned_to_staff = staff.sys_id;
            inc.status = 'Assigned';
            inc.update();
        }
    }
})(current, previous);

Job 1 — Monitor SLA Breaches
  Run    : Every 15 minutes (recommended)
  Note   : Field is current_time (NOT u_current_time — updated)
────────────────────────────────────────────────────────────────────────────────

(function scheduledJob() {
    var sla = new GlideRecord('x_1939650_smart_0_sla_metrics');
    sla.addQuery('u_sla_status', 'On Track');
    sla.query();
    while (sla.next()) {
        var now = new GlideDateTime();
        // Updated field: current_time (was u_current_time)
        sla.current_time = now; // stamp current check time
        var expRes = sla.u_expected_resolution.toString();
        if (expRes != '' && now.compareTo(new GlideDateTime(expRes)) > 0) {
            sla.u_sla_status = 'Breached';
            sla.u_escalation_level = parseInt(sla.u_escalation_level) + 1;
            sla.update();
        }
    }
})();


────────────────────────────────────────────────────────────────────────────────
Job 2 — Auto-Close Resolved Incidents
  Run    : Daily
────────────────────────────────────────────────────────────────────────────────

(function scheduledJob() {
    var cutoff = new GlideDateTime();
    cutoff.addSeconds(-86400);
    var inc = new GlideRecord('x_1939650_smart_0_guest_incidents');
    inc.addQuery('status', 'Resolved');
    inc.addQuery('resolved_date', '<', cutoff.getValue());
    inc.query();
    while (inc.next()) {
        inc.status = 'Closed';
        inc.update();
    }
})();


────────────────────────────────────────────────────────────────────────────────
Job 3 — Process Pending Notifications
  Run    : Every 5 minutes (recommended)
────────────────────────────────────────────────────────────────────────────────

(function scheduledJob() {
    var notif = new GlideRecord('x_1939650_smart_0_notifications');
    notif.addQuery('status', 'pending');
    notif.query();
    while (notif.next()) {
        notif.status = 'Sent';
        notif.sent_date = new GlideDateTime();
        notif.update();
    }
})();
